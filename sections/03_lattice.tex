\begin{frame}{Lattice QCD in one picture}
  \begin{columns}[c]

      \begin{itemize}
        \item Euclidean space-time grid with spacing $a$.
        \item Quarks live on sites; gluons are \textbf{link variables} $U_\mu(x)\in SU(3)$.
        \item Path integral evaluated by Monte Carlo sampling of gauge configurations:
      \end{itemize}

        $\Rightarrow \mN = 1077.8(48)$ MeV, \quad $|P_1|=2\pi/L = 339.8$ MeV

\begin{frame}{A nucleon with a staple inside}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Three-point function: nucleon source, staple operator $\bar q(0)\,\Gamma\,\mathcal{U}\,q(\bvec)$ at time $\tau$, nucleon sink at momentum $\vect P$.
        \item The plateau of the ratio to the two-point function gives the matrix element:
      \end{itemize}
      \[ R_{\Gamma}(\vect P;\mathcal C^{\rm lat})\equiv\mathrm{plateau}\left[\frac{C^{\rm 3pt}_{\Gamma}(\tau,\vect P;\mathcal C^{\rm lat})}{C^{\rm 2pt}(t_{\rm snk}-t_{\rm src},\vect P)}\right] \]
      \begin{itemize}
        \item Isovector $u-d$: disconnected diagrams cancel.
        \item Staple: legs of length $\eta|v|$ along $v$, separation $\bvec$.
      \end{itemize}
    \end{column}
