\begin{frame}{The $x$-dependent Sivers shift: Fourier representation}
  \centering
  \[\resizebox{0.72\textwidth}{!}{$\displaystyle \langle \vect{k}_y \rangle_{TU}(\vprp{\bvec}^2,x,\zetahat,\eta v \tcdot P)
	\ \equiv\ \mN \frac{\tilde f_{1T}^{\perp(1)}(\vprp{\bvec}^2,x;\zetahat,\ldots,\eta v \tcdot P)}{\tilde f_1^{(0)}(\vprp{\bvec}^2,x;\zetahat,\ldots,\eta v \tcdot P)}  $}\]
  \[\resizebox{0.98\textwidth}{!}{$\displaystyle =-m_{N} \frac{\int d(b\cdot P)e^{ix(b\cdot P)} \tAmp_{12B}(\bvec^2,\bvec \tcdot P,(\bvec \tcdot P) R(\zetahat^2)/\mN^2,-1/(\mN\zetahat)^2,\eta v \tcdot P)}{\int d(b\cdot P)e^{ix(b\cdot P)}\tAmp_{2B}(\bvec^2,\bvec \tcdot P,(\bvec \tcdot P) R(\zetahat^2)/\mN^2,-1/(\mN\zetahat)^2,\eta v \tcdot P)}  $}\]
  \[\resizebox{0.68\textwidth}{!}{$\displaystyle =-m_{N} \frac{\int d(b\cdot P)e^{ix(b\cdot P)} \left[\tAmp_{12B}^{\text{Re}}(\cdots)+i\tAmp_{12B}^{\text{Im}}(\cdots)\right]}{\int d(b\cdot P)e^{ix(b\cdot P)}\left[\tAmp_{2B}^{\text{Re}}(\cdots)+i\tAmp_{2B}^{\text{Im}}(\cdots)\right]}$}\]
  \small
  The Fourier transform in $b\tcdot P$ resolves the momentum fraction $x$.
  Both $\tAmp_{2B}$ and $\tAmp_{12B}$ retain their real and imaginary components.
\end{frame}

\begin{frame}{The $x$-dependent Sivers shift: cosine and sine contributions}
  \centering
  \small
  \[\resizebox{0.95\textwidth}{!}{$\displaystyle \langle\vect{k}_y\rangle_{TU}=-m_{N} \frac{\int d(b\cdot P)\left[\cos{\left(x(b\cdot P)\right)}+i\sin{\left(x(b\cdot P)\right)}\right] \left[\tAmp_{12B}^{\text{Re}}(\cdots)+i\tAmp_{12B}^{\text{Im}}(\cdots)\right]}{\int d(b\cdot P)\left[\cos{\left(x(b\cdot P)\right)}+i\sin{\left(x(b\cdot P)\right)}\right]\left[\tAmp_{2B}^{\text{Re}}(\cdots)+i\tAmp_{2B}^{\text{Im}}(\cdots)\right]}$}\]
  \[\resizebox{0.95\textwidth}{!}{$\displaystyle =-m_{N} \frac{\int d(b\cdot P) \left[\cos{\left(x(b\cdot P)\right)}\tAmp_{12B}^{\text{Re}}(\cdots)-\sin{\left(x(b\cdot P)\right)}\tAmp_{12B}^{\text{Im}}(\cdots)\right]}{\int d(b\cdot P)\left[\cos{\left(x(b\cdot P)\right)}\tAmp_{2B}^{\text{Re}}(\cdots)-\sin{\left(x(b\cdot P)\right)}\tAmp_{2B}^{\text{Im}}(\cdots)\right]} $}\]
  \[\resizebox{0.84\textwidth}{!}{$\displaystyle \boxed{\langle\vect{k}_y\rangle_{TU}=-m_{N} \frac{\left[\tAmp_{12B}^{[x]\text{Re}}(b^2,x,\zetahat,\eta v \tcdot P)-\tAmp_{12B}^{[x]\text{Im}}(b^2,x,\zetahat,\eta v \tcdot P)\right]}{\left[\tAmp_{2B}^{[x]\text{Re}}(b^2,x,\zetahat,\eta v \tcdot P)-\tAmp_{2B}^{[x]\text{Im}}(b^2,x,\zetahat,\eta v \tcdot P)\right]}}$}\]
  \footnotesize
  $\tAmp^{\text{Re}}$ is even and $\tAmp^{\text{Im}}$ is odd in $b\tcdot P$:
  the imaginary integrand is odd and integrates to zero on the symmetric domain.
  \[
    \tAmp_{iB}^{[x]\text{Re}}\equiv\int\frac{d(b\tcdot P)}{2\pi}\cos\!\big(x(b\tcdot P)\big)\tAmp_{iB}^{\text{Re}}(\cdots),\qquad
    \tAmp_{iB}^{[x]\text{Im}}\equiv\int\frac{d(b\tcdot P)}{2\pi}\sin\!\big(x(b\tcdot P)\big)\tAmp_{iB}^{\text{Im}}(\cdots).
  \]
\end{frame}

\begin{frame}{The Collins--Soper parameter $\zetahat$}
  \quick
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \[ \zetahat = \frac{v \cdot P}{\sqrt{|v^2|}\, \sqrt{P^2}} \]
      \begin{itemize}
        \item On the lattice the staple direction $v$ must be space-like ($v^2<0$), purely spatial ($v_0=0$).

        \item We track the Sivers shift over several $\zetahat$ and \textbf{extrapolate}.

\begin{frame}{Kinematic constraint to scan the full $\bvec\tcdot P$}
  \begin{itemize}
    \item The Fourier transform needs both zero and non-zero $\bvec \tcdot P$, but $\dfrac{v \tcdot \bvec}{v \tcdot P} = \bvec \tcdot P\, \dfrac{R(\zetahat^2)}{\mN^2}$ ties the geometry together. With $P = (E, P_1, 0, 0)$:
  \end{itemize}
  \[ b_{1}|v|\left[\frac{P_{1}}{\mN}\left(\zetahat-\sqrt{\zetahat^2+1}\right)+\zetahat\,\frac{\mN}{P_{1}}\right]=\vec{v}_{T}\cdot\vec{b}_{T} \]
  \begin{itemize}
    \item Choose $\vec{v}_{T}=(0,v_{3})$, $\vec{b}_{T}=(b_{2},0)$ $\Rightarrow$ right-hand side $=0$ for every $b_1$, so the bracket must vanish.
  \end{itemize}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{block}{Example: $P_1=-1$}
        \small
        $\mN = 0.6228(28)/a = 1077.8(48)$ MeV\\
        $P_1=-2\pi/L=-339.8$ MeV
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \small
      \[ \zetahat =\sqrt{\frac{P_{1}^2}{\mN^{2}\left(\frac{\mN^{2}}{P_{1}^{2}}+2\right)}} = \pm 0.090772 \]
      \[ \zetahat=\frac{-v_{1}P_{1}}{|v|\,\mN} \ \Rightarrow\ v_{1}=\pm 0.301\,v_{3} \]
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Symmetry improvement: path averaging for $v$}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Lattice links lie along the axes: several step-wise permutations end at the \textbf{same} lattice site.
        \item They are equivalent at the endpoints but carry different discretization artifacts and gauge noise.
        \item \textbf{Symmetry improvement:} average the correlators over all degenerate paths to the same site, before interpolating.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering

      {\scriptsize Two degenerate paths (solid, dashed) to the same site are averaged.}
    \end{column}
  \end{columns}
\end{frame}

% Slide 31
\begin{frame}{Real and imaginary amplitudes from $\pm\vect b$}
  \begin{itemize}
    \item Form symmetric and antisymmetric combinations of the two measured systems:
  \end{itemize}
  \[
    R^{[b^1\text{-even}]}=\tfrac12\big(R^{[+b^1]}+R^{[-b^1]}\big),\qquad
    R^{[b^1\text{-odd}]}=\tfrac12\big(R^{[+b^1]}-R^{[-b^1]}\big).
  \]
  \[
    \Gamma(1)=[+i\gamma^1],\quad \Gamma(2)=[-i\gamma^2],\quad
    \Gamma(4)=[+i\gamma^3],\quad \Gamma(8)=[\gamma^0].
  \]
  \begin{itemize}
    \item Each projection gives eight equations: real and imaginary parts of four Dirac structures.
    \item The even system isolates the real parts of $\tilde a_2,\tilde b_1,\tilde a_{12},\tilde b_8$; the odd system isolates their imaginary parts. Some rows also contain $b^1$-weighted cross terms.
  \end{itemize}
  \begin{block}{Convert the finite-link amplitudes}
    \[
      \tAmp_i=\tilde a_i,\qquad
      \tBmp_i=(\eta v\tcdot P)^n\,\tilde b_i,\qquad
      v\tcdot P=v^1|P^1|.
    \]
    $n$ is the power of $1/(v\tcdot P)$ multiplying $\tBmp_i$ in the parametrization; $E_N$ is the nucleon energy.
  \end{block}
\end{frame}

\begin{frame}{Solving for the amplitudes}
  \footnotesize
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \textbf{$b$-even (real parts)}
      \begin{align*}
        \tAmp^{\text{Re}}_2 &= \tfrac{1}{2}\text{Re}\{ R_{\Gamma(8)}\}^{\text{even}} \\
        \tBmp^{\text{Re}}_1 &= \tfrac{E_{N}|P^1|v^1}{2\mN^2  v^3} \text{Im}\{ R_{\Gamma(4)}\}^{\text{even}} \\
        \tAmp^{\text{Re}}_{12} &= \tfrac{-1}{2\mN  b^{(2)}}\Big[\text{Re}\{ R_{\Gamma(1)}\}^{\text{even}}+\tfrac{v^3}{v^1}\text{Re}\{ R_{\Gamma(4)}\}^{\text{even}}\Big] \\
        \tBmp^{\text{Re}}_8 &= \tfrac{-E_N |P^1|}{ 2\mN^3 b^{(2)} }\Big[\text{Im}\{ R_{\Gamma(8)}\}^{\text{even}}+\tfrac{|P^1|v^3}{E_N v^1}\text{Re}\{ R_{\Gamma(4)}\}^{\text{even}} \\ &\qquad + 2\mN  \tfrac{|P^1|}{E_N} b^{(2)}  \tAmp^{\text{Re}}_{12}\Big]
      \end{align*}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{$b$-odd (imaginary parts)}
      \begin{align*}
        \tAmp^{\text{Im}}_2 &= \tfrac{1}{2}\text{Im}\{ R_{\Gamma(8)}\}^{\text{odd}} \\
        \tBmp^{\text{Im}}_1 &= \tfrac{-E_{N}|P^1|v^1}{2\mN^2 v^3} \text{Re}\{ R_{\Gamma(4)}\}^{\text{odd}} \\
        \tAmp^{\text{Im}}_{12} &= \tfrac{-1}{2\mN  b^{(2)}}\Big[\text{Im}\{ R_{\Gamma(1)}\}^{\text{odd}}+\tfrac{v^3}{v^1}\text{Im}\{ R_{\Gamma(4)}\}^{\text{odd}}\Big] \\
        \tBmp^{\text{Im}}_8 &= \tfrac{E_N|P^1|}{ 2\mN^3 b^{(2)}  }\Big[\text{Re}\{ R_{\Gamma(8)}\}^{\text{odd}}-\tfrac{|P^1|v^3}{E_N v^1}\text{Im}\{ R_{\Gamma(4)}\}^{\text{odd}} \\ &\qquad - 2\mN  \tfrac{|P^1|}{E_N} b^{(2)}  \tAmp^{\text{Im}}_{12}\Big]
      \end{align*}
    \end{column}
  \end{columns}
  \normalsize
  \begin{block}{}
    \centering
    $\tAmp_{2B}^{\text{Re/Im}} = \tAmp_2^{\text{Re/Im}} + R(\zetahat^2)\, \tBmp_1^{\text{Re/Im}}, \qquad
     \tAmp_{12B}^{\text{Re/Im}} = \tAmp_{12}^{\text{Re/Im}} - R(\zetahat^2)\, \tBmp_8^{\text{Re/Im}}$\\[1mm]
    {\scriptsize $b^{(1)}, b^{(2)}$: components of $\vect b$ along $\hat 1$ (longitudinal) and $\hat 2$ (transverse).}\\
    then map $b_L = \dfrac{P\cdot b}{-P_L}$, \ $\bvec^2=b_L^2+b_T^2$: \quad $\tAmp_{iB}(b_L, b_T, \zetahat)\ \to\ \tAmp_{iB}(P \cdot b, \bvec^2, \zetahat)$
  \end{block}
\end{frame}
