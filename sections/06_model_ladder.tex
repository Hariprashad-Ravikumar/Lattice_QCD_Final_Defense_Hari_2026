\begin{frame}{Key idea 3: the Fourier transform picks the model}

      \textbf{A model must}
      \begin{itemize}
        \item describe all data with $|\vect b|\ge 3a$, $\eta\in[6,10]$;

% Slide 20
\begin{frame}{Rung 1: the Gaussian fails}
  \begin{columns}[c]
    \begin{column}{0.46\textwidth}
      \[ \tAmp_{2B}^{\text{Re}} = \gamma\, e^{-\alpha b_L^2-\beta b_T^2} \]\vspace{-2mm}\[ = \gamma \exp\!\left[-\frac{\alpha-\beta}{P_{L}^2} (P\cdot b)^{2}-\beta\, b^{2}\right] \]
      \begin{itemize}
        \item The Fourier transform over $P\cdot b$ needs $\alpha-\beta>0$.
        \item Fit at $P_L=-1$: \ $\alpha-\beta = -0.0427(27)$, \ $\chi^2/\text{dof}\approx 235$.
      \end{itemize}
      \keymsg{The data are \textbf{wider along $\vect P$ than across it}: the Gaussian \textbf{grows} in $P\cdot b$, so there is no Fourier transform.}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth]{Gaussian_A2B_Re_P1_FailedFit.pdf}\\
      {\scriptsize Gaussian fit to $\tAmp_{2B}^{\text{Re}}$, $P_L=-1$}
    \end{column}
  \end{columns}
\end{frame}

% Slide 21
\begin{frame}{Rung 2: let the data choose (physics-informed symbolic regression)}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \begin{itemize}
        \item \texttt{PySR} searches over formulas for $\tAmp_{2B}^{\text{Re}}$ ($P_L=-1$, $|\vect b|\ge3a$). Our loss adds a penalty when a formula grows at large $b_L$:
      \end{itemize}
      \[ \mathcal{L} = \frac{\chi^2}{\text{dof}} + \epsilon\, \mathcal{P}_{\text{asymptotic}} \]
      \begin{itemize}
        \item No Gaussian among the best formulas. A simple, accurate one:
      \end{itemize}
      \[ \tAmp_{2B}^{\text{Re}} = \frac{39.71}{(1 + b_L^2 + b_T^2)^{1.39}}\, e^{-0.534\eta} \]
      \begin{itemize}
        \item Caveat: these formulas depend on $b_L, b_T$ only through $b^2$. A \textbf{guide} to the form, not the final model.
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[width=\linewidth]{02_Gaussian_StraightLine_Test.pdf}\\
      {\scriptsize On $\log\tAmp$ vs $b^2$ a Gaussian is a straight line; the data bend upward.}
    \end{column}
  \end{columns}
\end{frame}

% Slide 22
\begin{frame}{Rung 3: exponential and separable forms fail}
  \begin{columns}[c]
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item $e^{-c|\vect b|}$ describes $\tAmp_{2B}^{\text{Re}}$ \textbf{alone} over $3\le|\vect b|/a\le8.2$ about as well as the power law.
        \item Separable forms (longitudinal profile in $P\cdot b$ $\times$ transverse profile), fit to all three amplitudes, lose the longitudinal decay in at least one amplitude or fit poorly.
        \item $e^{-c|P\cdot b|}$ has a cusp at $P\cdot b=0$, which conflicts with $\tAmp_{2B}^{\text{Im}}\propto b_L$.
      \end{itemize}
      {\scriptsize (Table of separable fits: backup)}
    \end{column}
    \begin{column}{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth]{03_Exponential_b_StraightLine_Test.pdf}
    \end{column}
  \end{columns}
\end{frame}

% Slide 23
\begin{frame}{Rung 4: a coupled power law}
  \begin{itemize}
    \item One denominator couples $(P\cdot b)^2$ and $b^2$: smooth at $P\cdot b=0$. The even amplitudes are absolutely integrable in $P\cdot b$ for $j>1/2$ (with a positive $b_L^2$ coefficient).
  \end{itemize}
  \[ \tAmp_{2B}^{\text{Re}}=\frac{a\, e^{-f \eta}}{\left[1 + \left(d + c\left(1 - \frac{k_1}{\eta} - \frac{k_2}{\eta^2}\right)\right) b_L^2 + d\, b_T^2\right]^j} \]
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\linewidth]{A2B_c_eta_AllAsymptoticModels_P1.pdf}\\
      {\scriptsize $c(\eta)$ for $\tAmp_{2B}^{\text{Re}}$ and candidate forms $c_1,\ldots,c_{10}$}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item $c(\eta)$ approaches an asymptote; we use $c_2(\eta)$ above: best AIC (fit quality per parameter), suppresses antiquarks.
        \item $d$ shows no $\eta$ dependence (backup).
        \item $\tAmp_{2B}^{\text{Im}}\propto b_L$ and $\tAmp_{12B}^{\text{Re}}$: same denominator form, with $\eta$-independent $c$, $d$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
