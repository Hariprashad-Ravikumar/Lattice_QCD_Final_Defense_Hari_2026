\begin{frame}{Extrapolating the parameters, not the curves}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Extrapolate each of the 12 model parameters $\{a,c,d,j\}\times$3 amplitudes ($c$ rescaled by $1/P_L^2$, $a_{\text{im}}$ by $1/P_L$): \ $p(\zetahat) = c + d/\zetahat^2$.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Use $P_L=-2,-3,-4$. $P_L=-1$ sits at $1/\zetahat=11.02$, far from the others ($3.40,\,1.85,\,1.27$).
        \item $c+d/\zetahat^2$ gave a lower $\chi^2/\text{dof}$ than $c+d/\zetahat$.
      \end{itemize}
    \end{column}
  \end{columns}
