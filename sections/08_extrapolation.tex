\begin{frame}{Staple dependence in the extrapolated reconstruction}
The global models contain a common factor and a longitudinal correction,
\[
 e^{-f\eta},\qquad
 c(\eta)=c\left(1-\frac{k_1}{\eta}-\frac{k_2}{\eta^2}\right),
 \qquad \lim_{\eta\to\infty}c(\eta)=c.
\]
\begin{itemize}
\item The reconstruction code uses $a,c,d,j$ for each amplitude. It uses the asymptotic coefficient $c$ and omits the common $e^{-f\eta}$ factor.
\item That common factor cancels in the Sivers ratio. No extrapolation of $f,k_1,k_2$ is needed for this asymptotic ratio within the selected model.
\item Individual amplitude plots therefore show spatial profiles with the common staple suppression removed. They are unrenormalized, and are not the literal $\eta\to\infty$ limits of $e^{-f\eta}$ times the profiles.
\end{itemize}
\figsrc{Thesis global ansatz; analysis notebook Extrapolation\_para\_level.ipynb, reconstruction of the twelve extrapolated parameters.}
\end{frame}

\begin{frame}{Real Fourier quantities and the reconstructed shift}
At fixed invariant $b^2=\rho^2$, define real quantities
\begin{align*}
 E(x,\rho)&=\int_{-\infty}^{\infty}\frac{d\nu}{2\pi}\,
 \cos(x\nu)\,\tAmp_{2B}^{\text{Re}}(\nu,\rho^2),\\
 O(x,\rho)&=\int_{-\infty}^{\infty}\frac{d\nu}{2\pi}\,
 \sin(x\nu)\,\tAmp_{2B}^{\text{Im}}(\nu,\rho^2),\\
 S(x,\rho)&=\int_{-\infty}^{\infty}\frac{d\nu}{2\pi}\,
 \cos(x\nu)\,\tAmp_{12B}^{\text{Re}}(\nu,\rho^2).
\end{align*}
\[
 \tilde f_1^{(0)}=2(E-O),\qquad
 \tilde f_{1T}^{\perp(1)}=-2S,\qquad
 \langle k_y\rangle_{TU}=-m_N\frac{S}{E-O}.
\]
$E,S$ are even in $x$; $O$ is odd. Amplitude normalizations are unrenormalized.

\end{frame}

\begin{frame}{Factorization on the lattice data}
At finite momentum, test the hypothesis
\[
 \tAmp(\nu,b^2)=F(\nu)G(b^2),\qquad \nu=P\cdot b.
\]
\begin{align*}
 T(\nu,b^2)&=\frac{\tAmp(\nu,b^2)}{\tAmp(4,b^2)}
       \overset{\rm fact.}{=}\frac{F(\nu)}{F(4)},\\
 B(\nu,b^2)&=\frac{\tAmp(\nu,b^2)}{\tAmp(\nu,18)}
       \overset{\rm fact.}{=}\frac{G(b^2)}{G(18)}.
\end{align*}
\begin{itemize}
\item $T$ must be independent of $b^2$; $B$ must be independent of $\nu$, wherever the denominators are nonzero.
\item Form each ratio on each Jackknife sample, after averaging the two lattice $\pm b_T$ measurements.
\item The sparse geometry requires interpolation before these ratios can be compared.
\end{itemize}
\figsrc{Thesis, finite-momentum ratio test; $b^2$ values in lattice units in the ratios.}
\end{frame}

\begin{frame}{Factorization after Fourier transformation}
The second hypothesis concerns the reconstructed shift,
\[
 f(x,\rho)\equiv K(x,\rho)
       \overset{?}{=}g(x)h(\rho).
\]
\begin{itemize}
\item This is a different test from separability of $\tAmp(\nu,b^2)$ on the lattice.
\item A coupled power law naturally correlates $x$ and $\rho$: the Fourier-transform width changes with $1+d\rho^2$.
\item A ratio can cancel some of the correlations in its numerator and denominator. Parameter differences alone are not a proof of nonfactorization.
\item Test the full reconstructed functions numerically, on the same domain for every Jackknife sample.
\end{itemize}
\figsrc{Thesis, $x$--$|b_T|$ factorization tests; conclusions below apply to the fitted functions.}
\end{frame}

\begin{frame}{The complete power-law transform and its width}
For $D=1+d\rho^2>0$, $C>0$, and $j>1/2$, define
\[
 I_j(x;C,D)=\int_{-\infty}^{\infty}\frac{d\nu}{2\pi}
     \frac{e^{ix\nu}}{(D+C\nu^2)^j}.
\]
\[
 I_j(x;C,D)=\frac{1}{\sqrt{\pi C}\,\Gamma(j)}
 \left(\frac{|x|}{2\sqrt{CD}}\right)^{j-1/2}
 K_{j-1/2}\!\left(|x|\sqrt{D/C}\right),\qquad x\ne0.
\]
\[
 I_j(0;C,D)=\frac{\Gamma(j-1/2)}{2\sqrt{\pi C}\,\Gamma(j)}D^{1/2-j}.
\]
$D$ enters both the prefactor and the $x$-dependent Bessel argument. Thus a separable product is not automatic, even for one amplitude.
\figsrc{Exact transform of the thesis power-law ansatz, using the $1/(2\pi)$ Fourier convention; obtained with a Gamma integral and a Gaussian transform.}
\end{frame}

\begin{frame}{Odd-amplitude transform and the $x=0$ limit}
For an odd amplitude $a_\nu\nu(D+C\nu^2)^{-j}$ and $x\ne0$,
\begin{align*}
 O(x;C,D)&=-a_\nu\,\frac{d}{dx}I_j(x;C,D)\\
 &=\frac{a_\nu\,\operatorname{sgn}(x)\sqrt{D/C}}
 {\sqrt{\pi C}\,\Gamma(j)}
 \left(\frac{|x|}{2\sqrt{CD}}\right)^{j-1/2}
 K_{j-3/2}\!\left(|x|\sqrt{D/C}\right).
\end{align*}
\begin{itemize}
\item Here $a_\nu=-a_{\rm im}/P_L$. The odd numerator shifts the Bessel order by one; it is not the same transform as an even amplitude.
\item Absolute integrability of the odd amplitude requires $j>1$; for $1/2<j\le1$, the sine transform at nonzero $x$ is conditionally convergent.
\item The extrapolated $j_{\rm im}=0.9921(92)$ makes behavior near $x=0$ delicate. A symmetric value at $x=0$ need not equal the one-sided limit.
\end{itemize}
\figsrc{Exact derivative of the preceding transform. The extrapolated exponent is from the thesis.}
\end{frame}

\begin{frame}{Least-squares distance from a separable function}
For real $f(x,\rho)$, minimize
\[
 J[g,h]=\int dx\,d\rho\,[f(x,\rho)-g(x)h(\rho)]^2.
\]
Stationarity gives the full functional derivatives
\begin{align*}
 \frac{\delta J}{\delta g(x)}
 &=-2\int d\rho\,f(x,\rho)h(\rho)
       +2g(x)\int d\rho\,h^2(\rho)=0,\\
 \frac{\delta J}{\delta h(\rho)}
 &=-2\int dx\,f(x,\rho)g(x)
       +2h(\rho)\int dx\,g^2(x)=0.
\end{align*}
Hence
\[
 h(\rho)=\frac{\int dx\,f(x,\rho)g(x)}{\int dx\,g^2(x)},\qquad
 g(x)=\frac{\int d\rho\,f(x,\rho)h(\rho)}{\int d\rho\,h^2(\rho)}.
\]
\figsrc{Thesis, factorization metric; derivative factors restored explicitly.}
\end{frame}

\begin{frame}{Iteration and the normalized residual}
Define, at iteration $n$,
\[
 Q_n=\int dx\,g_n^2(x),\qquad
 H_n(\rho)=\frac{\int dx'\,f(x',\rho)g_n(x')}{Q_n}.
\]
The full self-consistent update is
\[
 g_{n+1}(x)=
 \frac{\displaystyle\int d\rho\,f(x,\rho)
      \frac{\int dx'\,f(x',\rho)g_n(x')}{Q_n}}
 {\displaystyle\int d\rho\,
      \left[\frac{\int dx'\,f(x',\rho)g_n(x')}{Q_n}\right]^2},
 \qquad g_0(x)=1.
\]
\[
 N=\int dx\,d\rho\,f^2(x,\rho),\qquad
 \epsilon=\sqrt{J_{\min}/N}.
\]
Iterate to convergence independently on every Jackknife sample. Normalize the displayed factors with $\operatorname{mean}[g^2]=1$; $g\to\lambda g$, $h\to h/\lambda$ leaves their product unchanged.
\figsrc{Thesis, self-consistent factorization iteration and relative-distance definition.}
\end{frame}

\begin{frame}{Equivalent singular-value definition on the grid}
For quadrature weights $w_i$ in $x$ and $v_j$ in $\rho$, form
\[
 M_{ij}=\sqrt{w_i}\,f(x_i,\rho_j)\sqrt{v_j},\qquad
 M=U\Sigma V^{T},\quad \sigma_1\ge\sigma_2\ge0.
\]
The optimal product corresponds to the leading singular mode:
\[
 M_{(1)}=\sigma_1u_1v_1^T,\qquad
 N=\sum_{k=1}^{r}\sigma_k^2,\qquad
 J_{\min}=\sum_{k=2}^{r}\sigma_k^2.
\]
\[
 \boxed{\epsilon=\sqrt{1-\frac{\sigma_1^2}{\sum_{k=1}^{r}\sigma_k^2}}}
\]
This is the relative $L_2$ reconstruction error, not the fraction of the sum of singular values. On a uniform grid the constant weights cancel in $\epsilon$.
\figsrc{Linear-algebra characterization of the thesis least-squares objective; the thesis reports its iterative evaluation.}
\end{frame}

\begin{frame}{Ratio tests of the reconstructed Sivers shift}
For $f(x,\rho)=K(x,\rho)$,
\begin{align*}
 R_b(x;\rho)&=\frac{f(x,\rho)}{f(x,\rho_{\rm ref})}
      \overset{\rm fact.}{=}\frac{h(\rho)}{h(\rho_{\rm ref})},\\
 R_x(\rho;x)&=\frac{f(x,\rho)}{f(x_{\rm ref},\rho)}
      \overset{\rm fact.}{=}\frac{g(x)}{g(x_{\rm ref})}.
\end{align*}
\begin{itemize}
\item Choose $\rho_{\rm ref}=0.34$ fm and $x_{\rm ref}=0.3$.
\item Exact factorization makes $R_b$ flat in $x$ and $R_x$ flat in $\rho$, wherever the reference denominators are nonzero.
\item Non-flat reconstructed ratios test the selected fit continuously; they do not supply new lattice measurements between the sampled points.
\end{itemize}

\end{frame}

\begin{frame}{Scope of the extrapolation and factorization results}
\begin{itemize}
\item The thesis workflow carries finite-momentum fits through all twelve parameter extrapolations, Fourier reconstruction, and factorization tests.
\item The selected extrapolated Sivers shift admits an approximate product $g(x)h(\rho)$ with a $5.7$--$7.6\%$ relative $L_2$ residual on the tested domains.
\item The finite-momentum fitted functions also give non-flat ratios; the pattern is not created solely by rapidity extrapolation.
\item The direct lattice-data test remains limited by sparse sampling and interpolation. Statistical bands do not cover the full model and extrapolation uncertainty.
\item Larger momenta and denser geometry are needed to test these conclusions without relying as strongly on the selected ansatz.
\end{itemize}
\figsrc{Thesis, Extrapolations to the Physical Limit; distinction between raw-data tests and reconstructed-model tests retained.}
\end{frame}
