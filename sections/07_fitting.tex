% ===================== Fitting methodologies (thesis Ch. 6) =====================
\section{Fitting}

% Fitting methodology 1/32
\begin{frame}{Fitting three amplitudes on the spatial grid}
\begin{itemize}
\item Fit $\tAmp_{2B}^{\text{Re}}$, $\tAmp_{2B}^{\text{Im}}$, and $\tAmp_{12B}^{\text{Re}}$; the imaginary Sivers amplitude is consistent with zero and excluded.
\item Begin with $\tAmp_{2B}^{\text{Re}}$: its smaller uncertainties constrain the functional form most strongly.
\item Work on the measured $(b_L,b_T,\eta)$ grid at each fixed momentum, then map the fitted functions to invariant variables.
\end{itemize}
\[
 b_L=\frac{P\cdot b}{-P_L},\qquad b^2=b_L^2+b_T^2,\qquad
 b_T^2=b^2-\frac{(P\cdot b)^2}{P_L^2}.
\]
\begin{block}{Requirements on the model}
Correct even/odd parity, a controlled large-separation tail, and a stable large-$\eta$ limit for the Sivers ratio.
\end{block}
\end{frame}

% Fitting methodology 2/32
\begin{frame}{Gaussian ansatz: the Fourier-integrability condition}
\[
\tAmp_{iB}^{\text{Re}}=\gamma\exp[-\alpha b_L^2-\beta b_T^2]
=\gamma\exp\!\left[-\frac{\alpha-\beta}{P_L^2}(P\cdot b)^2-\beta b^2\right].
\]
\begin{itemize}
\item Hermitian conjugation requires the real amplitudes to be even in $b_L$.
\item The Fourier integral varies $P\cdot b$ at fixed $b^2$; decay on the measured spatial grid alone is insufficient.
\end{itemize}
\[
\int_{-\infty}^{\infty}d(P\cdot b)\,e^{ix(P\cdot b)}\tAmp_{iB}^{\text{Re}}
\quad\text{requires}\quad\boxed{\alpha-\beta>0}.
\]
\keymsg{If $\alpha<\beta$, the mapped Gaussian grows at large $|P\cdot b|$.}
\end{frame}

% Fitting methodology 5/32
\begin{frame}{Global fitting and the asymptotic staple limit}
\[
\tAmp_{iB}=F_i(b_L,b_T,\eta),\qquad |\vect b|\ge3a,\qquad6\le\eta\le10.
\]
\begin{itemize}
\item Fit the full retained grid, rather than individual fixed-$b^2$ trends.
\item Earlier $x$-integrated Sivers ratios approach a plateau at large staple extent; here the goal is the $x$-dependent ratio.
\item Test a common exponential staple profile, while allowing the longitudinal width to retain finite-$\eta$ dependence.
\item A common multiplicative factor can cancel in the ratio even when the individual amplitudes have not plateaued.
\end{itemize}
\begin{block}{Two distinct limits}
First control the staple limit $\eta\to\infty$ at fixed $\hat\zeta$; the later results section treats $\hat\zeta\to\infty$.
\end{block}
\end{frame}

% Fitting methodology 7/32
\begin{frame}{PySR: weighted loss and large-separation checks}
\small
\[
\mathcal L_{\rm base}=\frac{1}{N-k}\sum_{i=1}^{N}
\left[\frac{y_i-F_i}{\sigma_i}\right]^2,\qquad
\mathcal L_{\rm custom}=\mathcal L_{\rm base}+\epsilon\mathcal P,\quad\epsilon=0.1.
\]
\[
\mathcal P_{\rm thesis}=\sum_{q\in Q}|F(q)|,\qquad
Q=\{(10,50,1),(10,80,1),(10,10,1),(8,20,1),(6,30,2)\}.
\]
\begin{itemize}
\item Each tuple is $(\eta,b_L/a,b_T/a)$: check predictions beyond the measured grid, up to $b_L=80a$.
\item The thesis's included Julia implementation uses $\sum_qF(q)^2$ and the fixed normalization $N-3$, rather than the schematic absolute-value penalty and $N-k$ above.
\item Invalid candidate evaluations are penalized or rejected.
\end{itemize}
\begin{block}{What this check establishes}
A finite set of large-separation evaluations discourages growth. It does not prove convergence of the Fourier integral over the entire real line.
\end{block}
\end{frame}

\begin{frame}{Candidate selection: recover the Mellin-moment result}
\small
\begin{align*}
\int dx\,\tilde f_1^{(0)}&=2\int dx\,[\tAmp_{2B}^{[x]\text{Re}}-\tAmp_{2B}^{[x]\text{Im}}],\\

\end{align*}

\boxed{\langle k_y\rangle_{TU}^{\text{Mellin}}

\begin{itemize}
\item Compare with the earlier $b\cdot P=0$ generalized Sivers shift, where the amplitude ratio directly gives the integrated observable.

\end{itemize}

% Fitting methodology 25/32
\begin{frame}{Why the common staple exponential cancels}
\small
\[
\langle k_y\rangle_{TU}=-m_N
\frac{\int d\nu\,\cos(x\nu)\tAmp_{12B}^{\text{Re}}}
{\int d\nu\,[\cos(x\nu)\tAmp_{2B}^{\text{Re}}-\sin(x\nu)\tAmp_{2B}^{\text{Im}}]},\qquad\nu=b\cdot P.
\]
\[
\tAmp_i=e^{-f\eta}G_i\quad\Longrightarrow\quad
\langle k_y\rangle_{TU}=-m_N
\frac{e^{-f\eta}\int d\nu\,\cos(x\nu)G_{12B}^{\text{Re}}}
{e^{-f\eta}\int d\nu\,[\cos(x\nu)G_{2B}^{\text{Re}}-\sin(x\nu)G_{2B}^{\text{Im}}]}.
\]
\begin{itemize}
\item At fixed $\eta$, $e^{-f\eta}$ is independent of the integration variable and cancels exactly.
\item A simultaneous fit enforces the same $f$ for all three amplitudes.
\item The longitudinal width $c(\eta)$ still depends on staple extent; canceling the exponential does not yet take $\eta\to\infty$.
\end{itemize}
\end{frame}

% Fitting methodology 26/32
\begin{frame}{Invariant-space integrands and the remaining staple limit}
\small
\[
\langle k_y\rangle_{TU}=m_N
\frac{\int d\nu\,\cos(x\nu)\,a'/D_{12B}^{\text{Re}}}
{\int d\nu\,[\cos(x\nu)\,a/D_{2B}^{\text{Re}}-\sin(x\nu)\,a''\nu/(-P_L)/D_{2B}^{\text{Im}}]}.
\]
\begin{align*}
D_{12B}^{\text{Re}}&=[1+c'\nu^2/P_L^2+d'b^2]^{j'},\\
D_{2B}^{\text{Re}}&=[1+c(1-k_1/\eta-k_2/\eta^2)\nu^2/P_L^2+db^2]^j,\\
D_{2B}^{\text{Im}}&=[1+c''\nu^2/P_L^2+d''b^2]^{j''}.
\end{align*}
\begin{itemize}
\item Now take $c(\eta)\to c$ for the asymptotic staple result.
\item With positive nonsingular denominators and nonzero positive longitudinal coefficients, the real tails are absolutely integrable for $j,j'>1/2$; the extra $\nu$ numerator requires $j''>1$ for absolute integrability of the imaginary amplitude.
\end{itemize}
\end{frame}

% Fitting methodology 27/32
\begin{frame}{Full Jackknife covariance and correlated fitting}
\[
\bar y_i=\frac1{N_{\rm JK}}\sum_{k=1}^{N_{\rm JK}}y_i^{(k)},\qquad
C_{ij}=\frac{N_{\rm JK}-1}{N_{\rm JK}}\sum_{k=1}^{N_{\rm JK}}
(y_i^{(k)}-\bar y_i)(y_j^{(k)}-\bar y_j).
\]
\[
\boxed{\chi^2(\mathbf p)=\sum_{i,j}[\bar y_i-F_i(\mathbf p)]C^{-1}_{ij}[\bar y_j-F_j(\mathbf p)]}
\]
\begin{itemize}
\item Stack points from all retained spatial coordinates, staple extents, and all three amplitudes.
\item The same gauge configurations create correlations both between coordinates and between amplitude components.
\item Retain the full covariance rather than treating those measurements as independent.
\item Use \texttt{lsqfit} for nonlinear fitting and \texttt{gvar} for correlated Gaussian quantities.
\end{itemize}
\end{frame}

% Fitting methodology 28/32
\begin{frame}{From correlated data to the uncertainty on the Sivers shift}
\begin{enumerate}
\item Assemble the three amplitudes on the common retained grid at each momentum: $|\vect b|\ge3a$, $6\le\eta\le10$.
\item Construct the Jackknife covariance and fit the three models jointly with a common $f$.
\item Preserve parameter correlations when mapping the model to $(b\cdot P,b^2)$ and taking the staple limit.
\item Evaluate the cosine and sine transforms, combine the unpolarized terms, and form the Sivers ratio.
\item Carry the resampling uncertainty through the derived distributions and ratio; report Jackknife statistical bands.
\end{enumerate}
\begin{block}{Checks before interpreting the result}
Fit quality, parameter stability, longitudinal decay, negative-$x$ behavior, and the Mellin-moment comparison must be assessed together.
\end{block}
\end{frame}
