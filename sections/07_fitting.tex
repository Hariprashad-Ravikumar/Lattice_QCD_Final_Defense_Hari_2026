% ===================== Fitting methodologies (thesis Ch. 6) =====================
\section{Fitting}

% Fitting methodology 1/32
\begin{frame}{Fitting three amplitudes on the spatial grid}
\begin{itemize}
\item Fit $\tAmp_{2B}^{\text{Re}}$, $\tAmp_{2B}^{\text{Im}}$, and $\tAmp_{12B}^{\text{Re}}$; the imaginary Sivers amplitude is consistent with zero and excluded.
\item Begin with $\tAmp_{2B}^{\text{Re}}$: its smaller uncertainties constrain the functional form most strongly.
\item Work on the measured $(b_L,b_T,\eta)$ grid at each fixed momentum, then map the fitted functions to invariant variables.
\end{itemize}
\[
 b_L=\frac{P\cdot b}{-P_L},\qquad b^2=b_L^2+b_T^2,\qquad
 b_T^2=b^2-\frac{(P\cdot b)^2}{P_L^2}.
\]
\begin{block}{Requirements on the model}
Correct even/odd parity, a controlled large-separation tail, and a stable large-$\eta$ limit for the Sivers ratio.
\end{block}
\end{frame}

% Fitting methodology 2/32
\begin{frame}{Gaussian ansatz: the Fourier-integrability condition}
\[
\tAmp_{iB}^{\text{Re}}=\gamma\exp[-\alpha b_L^2-\beta b_T^2]
=\gamma\exp\!\left[-\frac{\alpha-\beta}{P_L^2}(P\cdot b)^2-\beta b^2\right].
\]
\begin{itemize}
\item Hermitian conjugation requires the real amplitudes to be even in $b_L$.
\item The Fourier integral varies $P\cdot b$ at fixed $b^2$; decay on the measured spatial grid alone is insufficient.
\end{itemize}
\[
\int_{-\infty}^{\infty}d(P\cdot b)\,e^{ix(P\cdot b)}\tAmp_{iB}^{\text{Re}}
\quad\text{requires}\quad\boxed{\alpha-\beta>0}.
\]
\keymsg{If $\alpha<\beta$, the mapped Gaussian grows at large $|P\cdot b|$.}
\end{frame}

% Fitting methodology 5/32
\begin{frame}{Global fitting and the asymptotic staple limit}
\[
\tAmp_{iB}=F_i(b_L,b_T,\eta),\qquad |\vect b|\ge3a,\qquad6\le\eta\le10.
\]
\begin{itemize}
\item Fit the full retained grid, rather than individual fixed-$b^2$ trends.
\item Earlier $x$-integrated Sivers ratios approach a plateau at large staple extent; here the goal is the $x$-dependent ratio.
\item Test a common exponential staple profile, while allowing the longitudinal width to retain finite-$\eta$ dependence.
\item A common multiplicative factor can cancel in the ratio even when the individual amplitudes have not plateaued.
\end{itemize}
\begin{block}{Two distinct limits}
First control the staple limit $\eta\to\infty$ at fixed $\hat\zeta$; the later results section treats $\hat\zeta\to\infty$.
\end{block}
\end{frame}

% Fitting methodology 7/32
\begin{frame}{PySR for $\tAmp_{2B}^{\text{Re}}$: weighted loss and large-separation checks}
\small
\begin{align*}
    \mathcal{L}_{\text{custom}} &= \frac{\chi^2}{\text{dof}} + \epsilon \sum_{b_T \in \{b_{T}^{\text{fixed}}\}} \sum_{b_L \in \{b_{L}^{\text{large}}\}} \left| \tAmp_{2B}^{\text{Re, model}}(b_L, b_T) \right|,\qquad \epsilon=0.1
\end{align*}
\begin{itemize}
\item The penalty evaluates each candidate at five test points $(\eta,b_L/a,b_T/a)=(10,50,1),(10,80,1),(10,10,1),(8,20,1),(6,30,2)$: predictions beyond the measured grid, up to $b_L=80a$.
\item Invalid candidate evaluations are penalized or rejected.
\end{itemize}
\begin{block}{What this check establishes}
A finite set of large-separation evaluations discourages growth. It does not prove convergence of the Fourier integral over the entire real line.
\end{block}
\end{frame}

\begin{frame}{Candidate selection: the negative-$x$ region}
\[
\tilde f_1^{(0)}(x)=2\left[\tAmp_{2B}^{[x]\text{Re}}(x)-\tAmp_{2B}^{[x]\text{Im}}(x)\right].

\begin{itemize}

\item This is a check on the combination of real and imaginary unpolarized amplitudes, not on either fit alone.
\end{itemize}
\[
\tilde f_1^{(0)}(x\lesssim-0.2)\longrightarrow0.
\]
\keymsg{The chosen $c_2(\eta)$ combines favorable fit criteria with the required behavior of the transformed distribution.}
\end{frame}

\begin{frame}{Candidate selection: recover the Mellin-moment result}
\small
\begin{align*}
\int dx\,\tilde f_1^{(0)}&=2\int dx\,[\tAmp_{2B}^{[x]\text{Re}}-\tAmp_{2B}^{[x]\text{Im}}],\\
\int dx\,\tilde f_{1T}^{\perp(1)}&=-2\int dx\,\tAmp_{12B}^{[x]\text{Re}}.
\end{align*}
\[
\boxed{\langle k_y\rangle_{TU}^{\text{Mellin}}
=m_N\frac{\int dx\,\tilde f_{1T}^{\perp(1)}}{\int dx\,\tilde f_1^{(0)}}}
\]
\begin{itemize}
\item Compare with the earlier $b\cdot P=0$ generalized Sivers shift, where the amplitude ratio directly gives the integrated observable.
\item The thesis tabulates integrations over $(-\infty,\infty)$, $[-1,1]$, $[-1,0]$, and $[0,1]$ to check support and contributions.
\item This tests all three fitted amplitudes. The observable is the ratio of integrated distributions, not the integral of the $x$-dependent ratio.
\end{itemize}
\end{frame}

\begin{frame}{Selected models for the three amplitudes}
\begin{align*}
\tAmp_{2B}^{\text{Re}}&=\frac{a\,e^{-f\eta}}{\left[1+\left(d+c\left(1-\frac{k_1}{\eta}-\frac{k_2}{\eta^2}\right)\right)b_L^2+db_T^2\right]^j},\\[2mm]
\tAmp_{12B}^{\text{Re}}&=\frac{-a'e^{-f\eta}}{[1+(c'+d')b_L^2+d'b_T^2]^{j'}},\\[2mm]
\tAmp_{2B}^{\text{Im}}&=\frac{a''b_Le^{-f\eta}}{[1+(c''+d'')b_L^2+d''b_T^2]^{j''}}.
\end{align*}
\begin{itemize}
\item The real amplitudes are even in $b_L$; the imaginary amplitude is odd.
\item Share $f$ across the three models, while fitting their spatial parameters separately.
\end{itemize}
\[
\langle \vect{k}_y \rangle_{TU} (x,b^2)= -\mN \frac{\int d(b\cdot P) \cos\!\left(x(b\cdot P)\right)\tAmp_{12B}^{\text{Re}}(\cdots)}{\int d(b\cdot P)\left[\cos\!\left(x(b\cdot P)\right)\tAmp_{2B}^{\text{Re}}(\cdots)-\sin\!\left(x(b\cdot P)\right)\tAmp_{2B}^{\text{Im}}(\cdots)\right]}\ .
\]
\end{frame}
