\begin{frame}{Path integral, Wick contractions and Monte Carlo}
  \quick
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \small
      Physical observables $\mathcal{O}$ are evaluated as an expectation value over the relevant degrees of freedom
      \begin{align*}
        \boxed{\langle \mathcal{O} \rangle =\frac{\int \mathcal{D}[U]\textcolor{blue}{\mathcal{D}[\psi]\mathcal{D}[\bar{\psi}]}~\mathcal{O}~e^{-[S_{\rm gauge}+\textcolor{blue}{\int dx\, \bar{\psi}\mathcal{M}\psi}]}}{\int\mathcal{D}[U]\textcolor{blue}{\mathcal{D}[\psi]\mathcal{D}[\bar{\psi}]}\,e^{-[S_{\rm gauge}+\textcolor{blue}{\int dx\, \bar{\psi}\mathcal{M}\psi}]}}}.
      \end{align*}
      The quark fields $\psi$ \& $\bar{\psi}$ are Grassmann variables:
      \begin{align*}
        \boxed{\langle \mathcal{O} \rangle =\frac{\int \mathcal{D}[U]~\widetilde{\mathcal{O}}~\textcolor{blue}{\det \mathcal{M}^2}\,e^{-S_{\rm gauge}}}{\int\mathcal{D}[U]\,\textcolor{blue}{\det \mathcal{M}^2}\,e^{-S_{\rm gauge}}}}.
      \end{align*}
      This integration results in the ``contraction'' of fermion--anti-fermion pairs in all possible ways (Wick's theorem), replacing them with quark propagators \textcolor{blue}{$\mathcal{M}^{-1}$}.
      \begin{itemize}
        \item Monte Carlo: sample configurations $U_n$ with probability $\propto \prod_f\det\mathcal M_f\; e^{-S_G[U]}$:
      \end{itemize}
      \[ \langle \mathcal O\rangle \approx \frac{1}{N}\sum_{n=1}^{N} \widetilde{\mathcal O}[U_n] \]
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \includegraphics[width=\linewidth]{mcstepsz2.pdf}\\
      {\scriptsize Markov-chain Monte Carlo history (illustration: $\mathbb{Z}_2$ gauge theory in 1+1D)}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Nucleon two- and three-point functions}
  \small
  \begin{align*}
    \chi_{\alpha}(\mathbf{x},t)&=\epsilon^{ijk}u_{\alpha}^{i}u^{j}_{\gamma}[C^{-1}\gamma_{5}]_{\gamma\delta}d^{k}_{\delta}\ (\mathbf{x},t)\\
    C_{\alpha\beta}(t,\mathbf{p})&=a^{3}\sum_{\mathbf{x}}e^{-i\mathbf{p}\cdot\mathbf{x}}\bra{0}\chi_{\alpha}(\mathbf{x},t)\bar{\chi}_{\beta}(\mathbf{0},0)\ket{0},\quad C_{\Gamma^+}(t,\mathbf{p})=\Gamma^{+}_{\beta\alpha}C_{\alpha\beta}\ \xrightarrow{t\to\infty}\ C\,e^{-Et}\\
    G_\Gamma(t,\tau;\mathbf{p})&=a^6\sum_{\mathbf{x}_2,\mathbf{x}_1}e^{-i\mathbf{p}\cdot\mathbf{x}_2}\,\Gamma^{\rm 3pt}_{\beta\alpha}\,\big\langle\chi_\alpha(\mathbf{x}_2,t)\,\mathcal O_\Gamma(\mathbf{x}_1,\tau)\,\bar\chi_\beta(\mathbf{0},0)\big\rangle,\quad \mathcal O_\Gamma=\bar q\,\Gamma\,q
  \end{align*}
  \vspace{-2mm}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item After integrating out the quarks, both are built from $\mathcal M^{-1}_u,\mathcal M^{-1}_u,\mathcal M^{-1}_d$; in $u-d$ the disconnected insertions cancel.
        \item The ratio at equal momentum removes the overlap $|Z|$ and the time dependence ($K(p)=-i\not\!p+m$, $\langle N|\mathcal O_\Gamma|N\rangle=\bar u\,\mathcal J\,u$):
      \end{itemize}
      \[
        R(t,\tau)=\frac{G_\Gamma(t,\tau;\mathbf p)}{C_{\Gamma^+}(t,\mathbf p)}\ \longrightarrow\ \frac{\mathrm{Tr}[\Gamma^{\rm 3pt}K(p)\,\mathcal J\,K(p)]}{4E(E+m)}\quad(\text{plateau in }\tau)
      \]
    \end{column}
