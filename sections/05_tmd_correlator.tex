\section{TMD correlator}

% Slide 14
\begin{frame}{Definition of TMDs}
  \begin{itemize}
    \item Consider a frame where the nucleon has large momentum in the $z$-direction, i.e., $P^+ \gg \mN$, $\vprp{P} = 0$.
      In light-cone coordinates, the components scale as \textcolor{brown}{\fbox{$\kei^+$:~$\vprp{\kei}$:~$\kei^-$ $\sim$ $P^+/\mN$:~$1$:~$\mN/P^+$}} under boosts along the $z$-axis.
    \item Gauge-invariant bilocal operator, divided by the soft factor $\widetilde{\softf}$:
  \end{itemize}
  \[ \Phi^{[\Gamma]}(\kei,P,S;\ldots)\equiv\int \frac{d^4 \bvec}{(2\pi)^4}\, e^{i \kei\cdot \bvec}\,
     \frac{\overbrace{\tfrac{1}{2}\, \bra{P,S}\ \bar{q}(0)\, \Gamma\ \WlineC{\mathcal{C}_\bvec}\ q(\bvec)\ \ket{P,S}}^{\textstyle \equiv\ \widetilde \Phi_\unsub^{[\Gamma]}(\bvec,P,S;\ldots)}}{\widetilde{\softf}(\bvec^2;\ldots)} \]
  \begin{itemize}
    \item Integrating over $k^-$ sets $\bvec^+=0$:
  \end{itemize}
  \[ \Phi^{[\Gamma]}(x,\vprp{k};P,S;\ldots)=\int \frac{d^2 \vprp{\bvec}}{(2\pi)^2} \int \frac{d (\bvec \tcdot P)}{(2\pi) P^+}\ e^{i x (\bvec \tcdot P) - i\vprp{\bvec} \tcdot \vprp{\kei}}\,\frac{\widetilde \Phi_\unsub^{[\Gamma]}}{\widetilde{\softf}}\Big|_{\bvec^+=0} \]
  \keymsg{$x \leftrightarrow \bvec \tcdot P$ \ and \ $\vprp{k} \leftrightarrow \vprp{b}$ are Fourier pairs. \qquad
  $\Phi^{[\gamma^+]} = f_1 - \dfrac{\myeps_{ij}\, k_{i}\, S_{j}}{\mN}\ f_{1T}^\prp$}
\end{frame}

% Slide 15
\begin{frame}{TMDs in Fourier space}
  \begin{itemize}
    \item The lattice computes $\bvec$-space matrix elements, so we state results for $\vprp b$-dependent TMDs:
  \end{itemize}
  \[ \tilde f(x, \vprp{\bvec}^2) \equiv \int  d^2 \vprp{\kei} \, e^{i \vprp{\bvec} \cdot \vprp{\kei} }\, f(x, \vprp{\kei}^2)
     = 2\pi \int d|\vprp{\kei}|\, |\vprp{\kei}|\ J_0(|\vprp{\bvec}||\vprp{\kei}|)\  f(x, \vprp{\kei}^2) \]
  \[ \tilde f^{(n)}(x, \vprp{\bvec}^2) \equiv n!\left( -\frac{2}{\mN^2}\partial_{\vprp{\bvec}^2} \right)^n \tilde f(x, \vprp{\bvec}^2)
     \ \xrightarrow{\ |\vprp b|\to0\ }\ \int  d^2 \vprp{\kei} \left( \frac{\vprp{\kei}^2}{2 \mN^2 }\right)^n  f(x, \vprp{\kei}^2) \]
  \begin{itemize}
    \item $\tilde f_1^{(0)}$: unpolarized; \ $\tilde f_{1T}^{\perp(1)}$: first $\vprp k$-moment of the Sivers function.
    \item $f_{1T}^{\perp(1)}(x)$ diverges without extra regularization (high-$\vprp k$ tail), so we \textbf{stay at finite $|\vprp b|$}.
    \item At finite $|\vprp b|$ the integrals converge, and $x$ is reached through $\bvec\tcdot P$.
  \end{itemize}
\end{frame}

\begin{frame}{Link geometry}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Staple-shaped link $\mathcal U[0,\eta v,\eta v+\bvec,\bvec]$: legs along $v$ of length $\eta|v|$, joined at the far end.
        \item $\eta\to\infty$: SIDIS ($\eta v\tcdot P\to+\infty$) or Drell--Yan ($\eta v\tcdot P\to-\infty$) staple.

        \item Light-like $v$ (the TMD limit) $\Leftrightarrow$ $\zetahat\to\infty$.

      \begin{enumerate}

\begin{frame}{Symmetry constraints: the amplitudes are complex}
  \begin{itemize}
    \item The correlator is decomposed into Lorentz-invariant amplitudes ($A_i$, $B_i$ in momentum space; $\tAmp_i$, $\tBmp_i$ in $\bvec$-space; next slide). Lorentz, parity, time reversal and Hermiticity constrain it; in momentum space Hermiticity makes $A_i$, $B_i$ \textbf{real}.
    \item In $\bvec$-space ($\kei\to -i\mN^2\bvec$), Hermitian conjugation gives
  \end{itemize}
  \[ \big[\widetilde{\Phi}^{[\GammaOp]}_{\unsub}(\bvec,P,S;v)\big]^* = \widetilde{\Phi}^{[\gamma^0 \GammaOp^\dagger \gamma^0]}_{\unsub}(-\bvec,P,S;v) \]
  \[ \Rightarrow\quad \big[\tAmp_i(\bvec^2,\bvec \tcdot P, \tfrac{v \tcdot \bvec}{v \tcdot P}, \zeta^{-2})\big]^* = \tAmp_i(\bvec^2,-\bvec \tcdot P, -\tfrac{v \tcdot \bvec}{v \tcdot P}, \zeta^{-2}) \]
  \keymsg{$\tAmp_i=\tAmp_i^{\text{Re}}+i\tAmp_i^{\text{Im}}$ \ are complex, with \ $\tAmp_i^{\text{Re}}$: even in $\bvec$, \quad $\tAmp_i^{\text{Im}}$: odd in $\bvec$ \quad (same for $\tBmp_i$)}
\end{frame}

\begin{frame}{Correlator parametrization}
  \footnotesize
  \begin{align*}
    \frac{1}{2}\widetilde \Phi^{[\gamma^\mu]}_{\unsub} ={}& P^\mu\, \tAmp_2 - i \mN^2 \bvec^\mu\, \tAmp_3 - i \mN \epsilon^{\mu \nu \alpha \beta} P_\nu \bvec_\alpha S_\beta\, \tAmp_{12} + \frac{\mN^2}{(v \tcdot P)} v^\mu\, \tBmp_1\\
    & + \frac{\mN}{v \tcdot P} \epsilon^{\mu \nu \alpha \beta} P_\nu v_\alpha S_\beta\, \tBmp_7 - \frac{i \mN^3}{v \tcdot P} \epsilon^{\mu \nu \alpha \beta} \bvec_\nu v_\alpha S_\beta\, \tBmp_8\\
    & - \frac{\mN^3}{v \tcdot P} (\bvec \tcdot S)\, \epsilon^{\mu \nu \alpha \beta} P_\nu \bvec_\alpha v_\beta\, \tBmp_9 - \frac{i \mN^3}{(v \tcdot P)^2} (v \tcdot S)\, \epsilon^{\mu \nu \alpha \beta} P_\nu \bvec_\alpha v_\beta\, \tBmp_{10}
  \end{align*}
  \normalsize
  \begin{itemize}
    \item The amplitudes depend on the invariants: \ $\tAmp_i\big(\bvec^2,\ \bvec \tcdot P,\ \tfrac{v\tcdot \bvec}{v\tcdot P},\ -\tfrac{1}{(\mN\zetahat)^2},\ \eta v \tcdot P\big)$
    \item In the TMD frame ($\bvec^+=0$, $\vprp v=\vprp P=0$) the third argument is fixed:
  \end{itemize}
  \[ \frac{v \tcdot \bvec}{v \tcdot P} = \bvec \tcdot P\, \frac{R(\zetahat^2)}{\mN^2}, \qquad R(\zetahat^2) \equiv 1- \sqrt{1 + \zetahat^{-2}} \]
  \begin{itemize}
    \item For $\Gamma=\gamma^+$ two combinations survive:
  \end{itemize}
  \[ \frac{1}{2 P^+}\widetilde \Phi^{[\gamma^+]}_{\unsub} = \tAmp_{2B} + i \mN \epsilon_{ij}\, \vect{\bvec}_i \vect{S}_j\, \tAmp_{12B}, \qquad
     \tAmp_{2B} \equiv \tAmp_2 + R(\zetahat^2) \tBmp_1, \quad \tAmp_{12B} \equiv \tAmp_{12} - R(\zetahat^2) \tBmp_8 \]
\end{frame}

\begin{frame}{Generalized Sivers shift from amplitudes}
  \begin{itemize}
    \item Average $k_y$ of unpolarized quarks in a nucleon polarized along $\hat x$ ($T$: transversely polarized nucleon, $U$: unpolarized quark), generalized to finite $|\vprp b|$:
  \end{itemize}
  \begin{block}{}
  \footnotesize
  \begin{align*}
    \langle \vect{k}_y \rangle_{TU}(\vprp{\bvec}^2,x,\zetahat,\eta v \tcdot P)
    &\equiv \mN \frac{\tilde f_{1T}^{\perp(1)}(\vprp{\bvec}^2,x;\zetahat,\ldots,\eta v \tcdot P)}{\tilde f_1^{(0)}(\vprp{\bvec}^2,x;\zetahat,\ldots,\eta v \tcdot P)}\\
    &=-\mN \frac{\int d(\bvec\tcdot P)\,e^{ix(\bvec\tcdot P)}\, \tAmp_{12B}\big(-\vprp{\bvec}^2,\bvec \tcdot P,(\bvec \tcdot P) R(\zetahat^2)/\mN^2,-1/(\mN\zetahat)^2,\eta v \tcdot P\big)}{\int d(\bvec\tcdot P)\,e^{ix(\bvec\tcdot P)}\,\tAmp_{2B}\big(-\vprp{\bvec}^2,\bvec \tcdot P,(\bvec \tcdot P) R(\zetahat^2)/\mN^2,-1/(\mN\zetahat)^2,\eta v \tcdot P\big)}
  \end{align*}
  \end{block}
  \begin{itemize}
    \item Under $\Gamma$-independent multiplicative renormalization, the soft factor and $Z$ cancel in this ratio at fixed geometry $(\bvec,P,\eta v)$.
    \item Setting $\bvec\tcdot P=0$ in the amplitudes, with no Fourier transform, gives the shift \textbf{integrated over $x$}: the quantity computed in earlier lattice work.
  \end{itemize}
  \keymsg{This thesis: scan the full $\bvec\tcdot P$ and Fourier transform, to get the shift \textbf{as a function of $x$}.}
\end{frame}
